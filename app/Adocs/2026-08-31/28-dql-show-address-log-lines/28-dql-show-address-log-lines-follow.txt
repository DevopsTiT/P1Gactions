SHOW ADDRESS LOG LINES (not counts)

fetch logs, from: now()-30m
| filter contains(content, "AddressKana")
| fields timestamp, dt.host_group.id, host.name, content
| limit 20

fetch logs, from: now()-30m
| filter contains(content, "address, Value =")
| fields timestamp, dt.host_group.id, host.name, content
| limit 20

fetch logs, from: now()-30m
| filter matchesRegex(content, "(?i)(AddressKana|AddressKanji|AXACleansingAddress|address\\s*,\\s*Value\\s*=|Type\\s*=\\s*address|PERSON_ADDRESS|OFFICE_ADDRESS)")
| fields timestamp, dt.host_group.id, host.name, content
| limit 30
